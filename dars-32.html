db.py
import sqlite3
from models import Mahsulot, Mijoz, Pozitsiya, Buyurtma, Holat


def ulan():
    con = sqlite3.connect(":memory:")
    con.execute("CREATE TABLE mahsulotlar (id INTEGER PRIMARY KEY, nom TEXT, narx INTEGER, qoldiq INTEGER, kategoriya TEXT)")
    con.execute("CREATE TABLE mijozlar (id INTEGER PRIMARY KEY, ism TEXT, xaridlar INTEGER DEFAULT 0)")
    con.execute("CREATE TABLE buyurtmalar (id INTEGER PRIMARY KEY, mijoz_id INTEGER, holat TEXT, jami INTEGER, sana TEXT)")
    con.execute("CREATE TABLE pozitsiyalar (id INTEGER PRIMARY KEY, buyurtma_id INTEGER, mahsulot_id INTEGER, soni INTEGER, narx INTEGER)")
    return con


def mahsulot_qosh(con, nom, narx, qoldiq, kategoriya="umumiy"):
    q = "INSERT INTO mahsulotlar (nom, narx, qoldiq, kategoriya) VALUES (?, ?, ?, ?)"
    return con.execute(q, (nom, narx, qoldiq, kategoriya)).lastrowid


def mahsulot_ol(con, id):
    q = con.execute("SELECT id, nom, narx, qoldiq, kategoriya FROM mahsulotlar WHERE id = ?", (id,)).fetchone()
    return Mahsulot(*q) if q else None


def mijoz_qosh(con, ism, xaridlar=0):
    return con.execute("INSERT INTO mijozlar (ism, xaridlar) VALUES (?, ?)", (ism, xaridlar)).lastrowid


def mijoz_ol(con, id):
    q = con.execute("SELECT id, ism, xaridlar FROM mijozlar WHERE id = ?", (id,)).fetchone()
    return Mijoz(*q) if q else None


def buyurtma_saqla(con, mijoz_id, pozitsiyalar, tolov, sana="2026-10-18"):
    bid = con.execute("INSERT INTO buyurtmalar (mijoz_id, holat, jami, sana) VALUES (?, ?, ?, ?)",
                      (mijoz_id, Holat.YANGI.value, tolov, sana)).lastrowid
    for p in pozitsiyalar:
        con.execute("INSERT INTO pozitsiyalar (buyurtma_id, mahsulot_id, soni, narx) VALUES (?, ?, ?, ?)",
                    (bid, p.mahsulot_id, p.soni, p.narx))
    return bid


def buyurtma_ol(con, id):
    q = con.execute("SELECT id, mijoz_id, holat, jami FROM buyurtmalar WHERE id = ?", (id,)).fetchone()
    if not q:
        return None
    poz = [Pozitsiya(*r) for r in con.execute(
        "SELECT mahsulot_id, soni, narx FROM pozitsiyalar WHERE buyurtma_id = ? ORDER BY id", (id,))]
    return Buyurtma(q[0], q[1], poz, Holat(q[2]), q[3])


def holat_yoz(con, id, holat):
    con.execute("UPDATE buyurtmalar SET holat = ? WHERE id = ?", (holat.value, id))

import db
from models import *

PROMO = {"YANGI10": 10, "BAHOR20": 20}


def sot(con, mahsulot_id, soni):
    if soni <= 0:
        raise ValueError("soni musbat bo'lsin")
    m = db.mahsulot_ol(con, mahsulot_id)
    if m is None:
        raise LookupError("mahsulot topilmadi")
    if m.qoldiq < soni:
        raise QoldiqYetarliEmas(m.nom + ": omborda yetarli emas")
    con.execute("UPDATE mahsulotlar SET qoldiq = qoldiq - ? WHERE id = ?", (soni, mahsulot_id))
    return m.qoldiq - soni


def chegirma(jami, promo=None, doimiy=False):
    if promo is not None:
        if promo not in PROMO:
            raise ValueError("noto'g'ri promo-kod")
        jami = jami * (100 - PROMO[promo]) / 100
    if doimiy:
        jami = jami * 0.95
    return round(jami)


RUXSAT = {
    Holat.YANGI: {Holat.TOLANDI, Holat.BEKOR},
    Holat.TOLANDI: {Holat.YETKAZILDI, Holat.BEKOR},
    Holat.YETKAZILDI: set(),
    Holat.BEKOR: set(),
}


def ozgartir(b, yangi):
    if yangi not in RUXSAT[b.holat]:
        raise ValueError("%s dan %s ga o'tib bo'lmaydi" % (b.holat.value, yangi.value))
    b.holat = yangi
    return b


def buyurtma_ber(con, mijoz_id, savat, promo=None):
    mijoz = db.mijoz_ol(con, mijoz_id)
    if mijoz is None:
        raise LookupError("mijoz topilmadi")
    try:
        pozitsiyalar = []
        for mid, soni in savat:
            m = db.mahsulot_ol(con, mid)
            sot(con, mid, soni)
            pozitsiyalar.append(Pozitsiya(mid, soni, m.narx))
        jami = sum(p.soni * p.narx for p in pozitsiyalar)
        tolov = chegirma(jami, promo, mijoz.xaridlar >= 5)
        bid = db.buyurtma_saqla(con, mijoz_id, pozitsiyalar, tolov)
        con.commit()
        return bid
    except Exception:
        con.rollback()
        raise


def tolov(con, buyurtma_id, summa):
    b = db.buyurtma_ol(con, buyurtma_id)
    if b is None:
        raise LookupError("buyurtma topilmadi")
    if summa != b.tolov:
        raise ValueError("to'lov summasi noto'g'ri")
    ozgartir(b, Holat.TOLANDI)
    db.holat_yoz(con, buyurtma_id, b.holat)
    con.execute("UPDATE mijozlar SET xaridlar = xaridlar + 1 WHERE id = ?", (b.mijoz_id,))
    con.commit()
    return True

import json


def zaxira(con, fayl):

                   for r in con.execute("SELECT id, nom, narx, qoldiq, kategoriya FROM mahsulotlar ORDER BY id")]
    mijozlar = [dict(zip(("id", "ism", "xaridlar"), r))
                for r in con.execute("SELECT id, ism, xaridlar FROM mijozlar ORDER BY id")]
    with open(fayl, "w", encoding="utf-8") as f:
        json.dump({"mahsulotlar": mahsulotlar, "mijozlar": mijozlar}, f, ensure_ascii=False, indent=2)
    return len(mahsulotlar)

    with open(fayl, encoding="utf-8") as f:

        con.execute("INSERT INTO mijozlar (id, ism, xaridlar) VALUES (?, ?, ?)", (m["id"], m["ism"], m["xaridlar"]))
