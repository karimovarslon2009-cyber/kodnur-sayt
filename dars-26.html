Bu dars yangi mavzu emas, loyiha. Sinflar, meros, dataclass, istisnolar va JSON'ni birlashtirib maktab kutubxonasi uchun tizim yozasiz: kitoblar, a'zolar, berish va qaytarish, jarima, tarix va faylga saqlash. Asosiy savol bitta: qaysi ish qaysi sinfga tegishli?

Bir nechta amaldan keyingi holat 🔮 Bashorat 🟡 O'rta

Soddalashtirilgan kutubxonada bir nechta amal ketma-ket bajariladi. Har birining natijasini kuzating.
🔮 Topshiriq:
  1. Kodni o'qing. band lug'atini va olgan sonini har amaldan keyin yozib boring.
  2. try / except qaysi chaqiruvlarda ishga tushishiga e'tibor bering.
  3. Dastur chiqaradigan qatorlarni yozing.
class Band(Exception):
    pass


band = {1: False, 2: False}
olgan = 0


def ber(id):
    global olgan
    if band[id]:
        raise Band("band")
    if olgan >= 2:
        raise ValueError("limit")
    band[id] = True
    olgan += 1


def qaytar(id):
    global olgan
    band[id] = False
    olgan -= 1


for amal, id in [("ber", 1), ("ber", 1), ("ber", 2), ("qaytar", 1), ("ber", 1)]:
    try:
        if amal == "ber":
            ber(id)
        else:
            qaytar(id)
        print(amal, id, "ok")
    except Band:
        print(amal, id, "band")
    except ValueError:
        print(amal, id, "limit")
print(band, olgan)

  4. qarzdorlar — hozir qo'lida kitobi bor va kun - ber_kun > 14 bo'lgan a'zolar ismlari, alifbo tartibida, takrorlanmasdan (kutubxona.berilgan dan oling).

kutubxona.py
from dataclasses import dataclass


class KutubxonaXato(Exception):
    pass


class KitobYoq(KutubxonaXato):
    pass


class BandKitob(KutubxonaXato):
    pass


class LimitXato(KutubxonaXato):
    pass


@dataclass
class Kitob:
    id: int
    nom: str
    muallif: str
    band: bool = False


class Azo:
    def __init__(self, ism):
        self.ism = ism
        self.kitoblar = []


class Kutubxona:
    def __init__(self):
        self.kitoblar = []
        self.azolar = {}
        self.berilgan = {}
        self.tarix = []

    def kitob_qosh(self, kitob):
        for k in self.kitoblar:
            if k.id == kitob.id:
                raise ValueError("Bu id band")
        self.kitoblar.append(kitob)

    def azo_qosh(self, ism):
        azo = Azo(ism)
        self.azolar[ism] = azo
        return azo

    def topish(self, id):
        for k in self.kitoblar:
            if k.id == id:
                return k
        raise KitobYoq("Kitob topilmadi: " + str(id))

    def qidir(self, matn):
        matn = matn.lower()
        return [k for k in self.kitoblar if matn in k.nom.lower() or matn in k.muallif.lower()]

    def ber(self, id, ism, kun):
        kitob = self.topish(id)
        if kitob.band:
            raise BandKitob(kitob.nom + " band")
        azo = self.azolar[ism]
        if len(azo.kitoblar) >= 3:
            raise LimitXato(ism + " 3 ta kitob olgan")
        kitob.band = True
        azo.kitoblar.append(id)
        self.berilgan[id] = (ism, kun)
        self.tarix.append(("ber", id, ism, kun))

    def qaytar(self, id, kun):
        if id not in self.berilgan:
            raise ValueError("Kitob berilmagan")
        ism, ber_kun = self.berilgan.pop(id)
        self.topish(id).band = False
        self.azolar[ism].kitoblar.remove(id)
        kechikish = max(kun - ber_kun - 14, 0)
        self.tarix.append(("qaytar", id, ism, kun))
        return kechikish * 500

    def to_dict(self):
        return {
            "kitoblar": [{"id": k.id, "nom": k.nom, "muallif": k.muallif, "band": k.band} for k in self.kitoblar],
            "azolar": {ism: a.kitoblar for ism, a in self.azolar.items()},
            "berilgan": {str(i): list(v) for i, v in self.berilgan.items()},
            "tarix": [list(t) for t in self.tarix],
        }

    @classmethod
    def from_dict(cls, d):
        k = cls()
        k.kitoblar = [Kitob(x["id"], x["nom"], x["muallif"], x["band"]) for x in d["kitoblar"]]
        for ism, ids in d["azolar"].items():
            k.azolar[ism] = Azo(ism)
            k.azolar[ism].kitoblar = list(ids)
        k.berilgan = {int(i): tuple(v) for i, v in d["berilgan"].items()}
        k.tarix = [tuple(t) for t in d["tarix"]]
        return k


import json


def saqla(kutubxona, fayl):
    with open(fayl, "w", encoding="utf-8") as f:
        json.dump(kutubxona.to_dict(), f, ensure_ascii=False, indent=2)


def yukla(fayl):
    with open(fayl, encoding="utf-8") as f:
        return Kutubxona.from_dict(json.load(f))

💡 Asosiy fikr: Loyihadagi xatolarning ko'pi holat bir joyda yangilanib, boshqa joyda unutilganidan chiqadi. Dastur yiqilmaydi, ma'lumot esa zid bo'lib qoladi.

⚠️ Qaytarishda a'zo ro'yxati unutilganKitob band emas, lekin a'zo hamon "unda". Limit noto'g'ri ishlaydi.
⚠️ Tekshiruvdan oldin o'zgartirishAvval band = True qilib, keyin limitni tekshirsangiz, rad etilgan kitob ham band bo'lib qoladi. Avval tekshiring, keyin o'zgartiring.
